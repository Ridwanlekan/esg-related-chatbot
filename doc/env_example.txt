# --- Voice mode (STT + TTS) ----------------------------------------------------
# The mic button sits next to the chat input. First press records, second press
# stops and transcribes into the composer; answers are then read back aloud.
# Requires HTTPS (or localhost) or the browser blocks getUserMedia.
#
# Both voice providers default to "local": STT via faster-whisper, TTS via
# Kokoro. Neither needs an Azure deployment, which is why this works out of the
# box; "openai" moves either one onto the Azure OpenAI endpoint above and then
# needs a deployment provisioned there.
#
# VOICE_STT_PROVIDER = "local"     # "local" = faster-whisper on this machine
#                                 # "openai" = Whisper API on the Azure OpenAI
#                                 #   endpoint above (no extra dependency)
# VOICE_STT_MODEL = "small"        # local only: tiny/base/small/medium/large
# VOICE_TTS_PROVIDER = "kokoro"    # "kokoro" = local Kokoro v1.0 (WAV out)
#                                 # "openai" = Azure speech endpoint (mp3 out)
# VOICE_TTS_VOICE = "af_heart"     # kokoro: af_* am_* bf_* bm_* (54 voices).
#                                 #   openai: alloy | nova | onyx | shimmer
# VOICE_TTS_MODEL = "gpt-4o-mini-tts"   # openai only: deployment name
# VOICE_TTS_MODEL_PATH = "models/tts/kokoro-v1.0.onnx"   # kokoro only
# VOICE_TTS_VOICES_PATH = "models/tts/voices-v1.0.bin"    # kokoro only
# VOICE_TTS_LANG = "en-gb"         # kokoro G2P language: en-us | en-gb | ...
# VOICE_TTS_SPEED = "1.0"          # kokoro playback rate
# VOICE_MAX_UPLOAD_MB = 5          # hard cap on an uploaded recording
# VOICE_TTS_MAX_CHARS = 4000       # cap on characters sent to TTS
# VOICE_STT_PRICE_PER_MIN = 0.006  # $/audio-minute, for the usage dashboard only
# VOICE_TTS_PRICE_PER_M = 0.60     # $/1M characters, ditto. Local TTS costs
#                                 #   nothing per request; set to 0 to see that.
#
# Both "local" providers need the extra: pip install 'esg-chatbot[voice]'.
# No ffmpeg binary and no espeak-ng system package are required — faster-whisper
# decodes through PyAV (which bundles ffmpeg) and Kokoro gets espeak-ng as a
# wheel via espeakng-loader. Without the extra, /voice/stt and /voice/tts return
# 503 with that message and the mic degrades to typing — the rest of the app is
# unaffected.
#
# Weights: the faster-whisper "small" model (~484 MB) downloads from HuggingFace
# on first use. Kokoro does not — its ~350 MB of weights are baked into the
# container image by the Dockerfile, and locally you fetch them once with
#   curl -L -o models/tts/kokoro-v1.0.onnx \
#     https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/kokoro-v1.0.onnx
#   curl -L -o models/tts/voices-v1.0.bin \
#     https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/voices-v1.0.bin
#
# Note: audio is held in memory for the length of the request and never written
# to disk or logged. Local STT passes the clip through a temp file because
# faster-whisper needs a seekable path; it is deleted as soon as it returns.
# DOCLING_ASR = 1                 # 0 disables audio/video transcription
# DOCLING_ASR_MODEL = WHISPER_TURBO   # any docling asr_model_specs name
# DOCLING_VET_DIR = .docling_vet  # cleaned Markdown saved before ingest (gitignored; 0/false disables)
# PORT = 8000                     # uvicorn port (platforms like App Service inject this)
# LOG_LEVEL = INFO

# --- Multilingual answers (Mandarin + Hindi) ----------------------------------
# Chat, voice and the UI can answer in English, Simplified Chinese (zh-Hans)
# and Hindi (hi). No configuration is required to enable them: a question typed
# in Chinese or Hindi is answered in that language, an English question follows
# the language picked in the UI selector, and English remains the default.
#
# Retrieval stays English-only on purpose. The embedding index and reranker are
# English-centric, so a Zh/Hi question is translated to English before it is
# embedded (one extra LLM call, metered under the "rewrite" kind) while the
# original question is kept for the answer, so the reply still comes back in
# the user's language.
#
# Voice follows the same rule: STT auto-detects the spoken language, and Kokoro
# TTS switches to a Mandarin or Hindi voice for those languages (ignoring an
# English VOICE_TTS_VOICE when it would mispronounce the text).
# SUPPORTED_LANGS = en,zh,hi      # informational; the set is fixed in code

# --- Organisations, billing & free tier ---------------------------------------
# The product is sold per organisation through Stripe. An organisation buys a
# plan that grants workspaces, seats and a monthly question pool; payment state
# is written by the signed Stripe webhook. See doc/solution-workflow.md for the
# end-to-end workflow.
#
# STRIPE_SECRET_KEY = ""          # Stripe API key (sk_...); required for checkout/portal/webhook
# STRIPE_WEBHOOK_SECRET = ""      # whsec_... verifies POST /billing/webhook signatures
# STRIPE_PORTAL_CONFIGURATION_ID = ""   # optional pinned portal config that enables plan
#                                 # switching; if unset and the account default already switches,
#                                 # that is used, otherwise one named "esg-plan-switcher" is created
#
# Base per-workspace prices — a tier is purchasable only when its price ids are set:
# STRIPE_PRICE_TEAM_MONTHLY = ""
# STRIPE_PRICE_TEAM_ANNUAL = ""
# STRIPE_PRICE_BUSINESS_MONTHLY = ""
# STRIPE_PRICE_BUSINESS_ANNUAL = ""
# STRIPE_PRICE_ENTERPRISE_ANNUAL = ""
#
# Metered add-ons (extra seats past the per-workspace allowance, question overage):
# STRIPE_PRICE_EXTRA_SEAT_TEAM = ""
# STRIPE_PRICE_EXTRA_SEAT_TEAM_ANNUAL = ""
# STRIPE_PRICE_EXTRA_SEAT_BUSINESS = ""
# STRIPE_PRICE_EXTRA_SEAT_BUSINESS_ANNUAL = ""
# STRIPE_PRICE_QUESTION_OVERAGE = ""
# STRIPE_PRICE_QUESTION_OVERAGE_ANNUAL = ""
#
# Free tier: self-registration lands in the shared sample organisation (_sample).
# MAX_FREE_ACCOUNTS = 500         # cap on free accounts; 0 closes self-signup (503 at the cap)
# FREE_TIER_QUESTION_CAP = 50     # answered questions per account; 0/negative = unlimited
# EMAIL_VERIFICATION_REQUIRED = 1 # 1 gates free accounts behind a confirmed email
# VERIFICATION_TOKEN_TTL_HOURS = 48   # lifetime of a confirmation link
#
# Locally, Stripe webhooks do not arrive without a forwarder. --api-key pins the
# CLI to the STRIPE_SECRET_KEY account (the CLI's own login/sandbox may differ),
# and recent CLIs require an explicit --events list:
#   stripe listen --api-key "$STRIPE_SECRET_KEY" \
#     --events checkout.session.completed,customer.subscription.created,customer.subscription.updated,customer.subscription.deleted,invoice.paid,invoice.payment_failed \
#     --forward-to localhost:8000/billing/webhook
# The listener prints "Your webhook signing secret is whsec_..."; copy that into
# STRIPE_WEBHOOK_SECRET. It is per-account, so a secret from a different account
# or sandbox makes every event fail signature verification (400) and nothing
# reflects. Without the forwarder, reconcile with
# POST /admin/organisations/{id}/billing/refresh.
